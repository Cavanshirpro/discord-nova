cmake_minimum_required(VERSION 3.25)
project(discord_nova VERSION 0.1.0 DESCRIPTION "Native C++ Discord client with GUI and TUI" LANGUAGES CXX)

option(NOVA_BUILD_GUI "Build the Qt Widgets desktop client" ON)
option(NOVA_BUILD_TUI "Build the original Nova terminal client" ON)
option(NOVA_NATIVE_MEDIA "Enable the original TUI's native voice/video backend (requires libdave, MLSPP, OpenSSL and Opus)" OFF)
option(NOVA_BUILD_TESTS "Build offline tests" ON)
option(NOVA_BROWSER_IDENTITY "Build Chrome TLS/HTTP profiles with the pinned curl-impersonate runtime" ON)
set(NOVA_DISCORD_SOURCE "" CACHE PATH "Use a local discord.cpp-self checkout instead of downloading")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(MSVC)
    add_compile_options(/utf-8)
endif()
include(FetchContent)
include(GNUInstallDirs)
find_package(Threads REQUIRED)
include(CheckCXXSourceCompiles)
check_cxx_source_compiles("#include <thread>\n#include <stop_token>\nint main() { std::jthread worker([] {}); }" NOVA_HAS_JTHREAD)
if(NOT NOVA_HAS_JTHREAD)
    message(FATAL_ERROR "C++20 std::jthread/stop_token support is required. On macOS, select Xcode 26 or newer.")
endif()

if(NOVA_BROWSER_IDENTITY)
    include(cmake/BrowserCurl.cmake)
else()
    set(DISCORD_SELF_CURL_IMPERSONATE OFF CACHE BOOL "" FORCE)
    if(CURL_DIR MATCHES "/browser-curl$")
        unset(CURL_DIR CACHE)
        unset(CURL_DIR)
    endif()
endif()

set(DISCORD_SELF_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(DISCORD_SELF_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(DISCORD_SELF_BUILD_LIVE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(DISCORD_SELF_NATIVE_MEDIA ${NOVA_NATIVE_MEDIA} CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if(NOVA_DISCORD_SOURCE)
    add_subdirectory("${NOVA_DISCORD_SOURCE}" "${CMAKE_BINARY_DIR}/discord-self" EXCLUDE_FROM_ALL)
else()
    FetchContent_Declare(discord_self
        GIT_REPOSITORY https://github.com/Cavanshirpro/discord.cpp-self.git
        GIT_TAG 0d37a2ebe160601a9d2e219004bcbe4b39b94c7f) # v0.2.2 + browser/media transport options
    FetchContent_MakeAvailable(discord_self)
endif()

if(WIN32)
    # Windows SDK macros otherwise expand std::min/max and numeric_limits::max
    # inside discord.cpp-self when curl pulls in the native Windows headers.
    target_compile_definitions(discord_cpp_self PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(NOVA_NATIVE_MEDIA)
    install(FILES cmake/licenses/libdave.txt cmake/licenses/MLSPP.txt
        DESTINATION share/licenses/native-media COMPONENT Runtime)
endif()

add_library(nova_core STATIC src/core.cpp)
target_include_directories(nova_core PUBLIC include)
target_link_libraries(nova_core PUBLIC discord::self Threads::Threads)
target_compile_definitions(nova_core PUBLIC NOVA_VERSION="${PROJECT_VERSION}")
if(WIN32)
    target_compile_definitions(nova_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
    target_compile_options(nova_core PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(nova_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(NOVA_BUILD_GUI)
    find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
    qt_add_executable(discord-nova-gui src/gui.cpp)
    qt_add_resources(discord-nova-gui nova_theme PREFIX "/nova" FILES styles/nova.qss styles/chevron.xpm)
    set_target_properties(discord-nova-gui PROPERTIES
        WIN32_EXECUTABLE TRUE MACOSX_BUNDLE TRUE
        MACOSX_BUNDLE_GUI_IDENTIFIER io.github.cavanshirpro.discord-nova
        MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
        MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
    target_link_libraries(discord-nova-gui PRIVATE nova_core Qt6::Widgets)
    if(NOVA_BROWSER_IDENTITY)
        nova_browser_runtime(discord-nova-gui)
    endif()
    install(TARGETS discord-nova-gui BUNDLE DESTINATION . COMPONENT Runtime RUNTIME DESTINATION bin COMPONENT Runtime)
endif()

if(NOVA_BUILD_TUI)
    add_executable(discord-nova-tui src/tui/main.cpp src/tui/backend.cpp src/tui/app.cpp)
    target_link_libraries(discord-nova-tui PRIVATE nova_core)
    if(NOT MSVC)
        target_compile_options(discord-nova-tui PRIVATE -O1 -Wall -Wextra -Wno-missing-field-initializers)
    endif()
    if(WIN32)
        target_link_libraries(discord-nova-tui PRIVATE comdlg32 shell32 user32)
    endif()
    if(MINGW)
        target_link_options(discord-nova-tui PRIVATE -municode)
    endif()
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        # The imported UI uses a large JSON-backed model; bound GCC's peak memory.
        target_compile_options(discord-nova-tui PRIVATE --param=ggc-min-expand=5 --param=ggc-min-heapsize=4096)
        set_source_files_properties(src/tui/app.cpp PROPERTIES COMPILE_OPTIONS "--param=ggc-min-expand=1;--param=ggc-min-heapsize=1024")
    endif()
    if(NOVA_BROWSER_IDENTITY)
        nova_browser_runtime(discord-nova-tui)
    endif()
    install(TARGETS discord-nova-tui RUNTIME DESTINATION bin COMPONENT Runtime)
endif()

if(NOVA_BUILD_TESTS)
    enable_testing()
    add_executable(nova_core_tests tests/core.cpp)
    target_link_libraries(nova_core_tests PRIVATE nova_core)
    if(NOVA_BROWSER_IDENTITY)
        nova_browser_runtime(nova_core_tests)
    endif()
    add_test(NAME core COMMAND nova_core_tests)
    find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
    add_executable(nova_integration_tests tests/integration.cpp)
    target_link_libraries(nova_integration_tests PRIVATE nova_core)
    if(NOVA_BROWSER_IDENTITY)
        nova_browser_runtime(nova_integration_tests)
    endif()
    add_test(NAME loopback COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/loopback.py $<TARGET_FILE:nova_integration_tests>)
    set_tests_properties(core loopback PROPERTIES TIMEOUT 45)
    if(NOVA_BROWSER_IDENTITY)
        add_executable(nova_browser_probe tests/browser_probe.cpp)
        target_link_libraries(nova_browser_probe PRIVATE nova_core)
        nova_browser_runtime(nova_browser_probe)
        add_test(NAME browser_tls COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/browser_tls.py $<TARGET_FILE:nova_browser_probe>)
        add_test(NAME browser_loopback COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/loopback.py $<TARGET_FILE:nova_integration_tests>
            chrome152 ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/localhost.pem ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/localhost-key.pem)
        set_tests_properties(browser_tls browser_loopback PROPERTIES TIMEOUT 45)
    endif()
    if(NOVA_BUILD_GUI)
        add_test(NAME gui_demo COMMAND discord-nova-gui --demo --smoke-test)
        set_tests_properties(gui_demo PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
    endif()
    if(NOVA_BUILD_TUI)
        if(NOVA_NATIVE_MEDIA)
            find_package(OpenSSL 3 REQUIRED)
            get_target_property(_nova_discord_dir discord_cpp_self SOURCE_DIR)
            add_executable(nova_media_transport_probe tests/media_transport_probe.cpp)
            target_include_directories(nova_media_transport_probe PRIVATE
                "${_nova_discord_dir}/src" "${DISCORD_SELF_DAVE_INCLUDE_DIR}" "${DISCORD_SELF_OPUS_INCLUDE_DIR}")
            set_property(TARGET nova_media_transport_probe PROPERTY NO_SYSTEM_FROM_IMPORTED TRUE)
            target_include_directories(nova_media_transport_probe BEFORE PRIVATE
                $<TARGET_PROPERTY:CURL::libcurl,INTERFACE_INCLUDE_DIRECTORIES>)
            target_link_libraries(nova_media_transport_probe PRIVATE nova_core CURL::libcurl OpenSSL::Crypto)
            if(NOVA_BROWSER_IDENTITY)
                nova_browser_runtime(nova_media_transport_probe)
            endif()
            add_test(NAME media_transport_tls COMMAND ${Python3_EXECUTABLE}
                ${CMAKE_CURRENT_SOURCE_DIR}/tests/media_transport_tls.py $<TARGET_FILE:nova_media_transport_probe> ${NOVA_BROWSER_IDENTITY})
            set_tests_properties(media_transport_tls PROPERTIES TIMEOUT 25)
        endif()
        add_executable(nova_terminal_input_tests tests/terminal_input.cpp)
        target_include_directories(nova_terminal_input_tests PRIVATE src/tui)
        add_test(NAME terminal_input COMMAND nova_terminal_input_tests)
        add_test(NAME tui_demo COMMAND discord-nova-tui --demo --smoke-test)
        add_test(NAME tui_model COMMAND discord-nova-tui --self-test)
        add_test(NAME tui_rest COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/tui_rest.py $<TARGET_FILE:discord-nova-tui>)
        set_tests_properties(tui_demo tui_model tui_rest PROPERTIES TIMEOUT 60)
        if(UNIX)
            add_test(NAME terminal_pty COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/terminal_pty.py $<TARGET_FILE:discord-nova-tui>)
            set_tests_properties(terminal_pty PROPERTIES TIMEOUT 25)
        endif()
        if(NOVA_BROWSER_IDENTITY)
            add_test(NAME tui_browser_rest COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/tui_rest.py $<TARGET_FILE:discord-nova-tui>
                chrome152 ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/localhost.pem ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/localhost-key.pem)
            set_tests_properties(tui_browser_rest PROPERTIES TIMEOUT 60)
        endif()
    endif()
endif()

install(FILES README.md LICENSE COPYING NOTICE THIRD_PARTY.md CHANGELOG.md ROADMAP.md
    CONTRIBUTING.md SECURITY.md release.md VERSION DESTINATION share/doc/discord-nova COMPONENT Runtime)
install(DIRECTORY docs/ DESTINATION share/doc/discord-nova/docs COMPONENT Runtime)
install(FILES config.example.json DESTINATION . COMPONENT Runtime)
if(WIN32)
    install(FILES start-tui.cmd start-demo.cmd DESTINATION . COMPONENT Runtime)
endif()
